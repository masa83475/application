scanmeter-app 現状構成・ソースコード集

このファイルは更新時点のプロジェクト構成とソースコードを一つにまとめたものです。
目的: 検針票PDFをOCRし、CSVとして出力するローカルWebアプリ。

【構成概要】
- Python / FastAPI によるローカルWebアプリ。トップ画面、PDFアップロード、進捗取得、CSVダウンロード、クリーンアップを提供。
- PDFページを画像化してTesseract日本語OCRを実行し、OCR結果をCSVとして出力。
- 電力会社判定、4社（東京電力・中部電力・北陸電力・東北電力）の抽出処理、項目別CSV生成コードも収録。
- runtime/ や temp/ の内容、キャッシュ、バイナリなどの生成物は収録していない。
- 下記の構成図は収録対象のソース・設定ファイルを中心に示す。

import os
import glob
import json

from fastapi import APIRouter

router = APIRouter()

from app.config.settings import TEMP_DIR

PROGRESS_FILE = TEMP_DIR / "progress.json"


@router.post("/cleanup")
async def cleanup():

    # CSV削除
    for file in glob.glob(
        os.path.join(
            TEMP_DIR,
            "*.csv"
        )
    ):

        try:
            os.remove(file)
        except Exception:
            pass

    # PDF削除
    for file in glob.glob(
        os.path.join(
            TEMP_DIR,
            "*.pdf"
        )
    ):

        try:
            os.remove(file)
        except Exception:
            pass

    # progress初期化
    try:

        with open(
            PROGRESS_FILE,
            "w",
            encoding="utf-8"
        ) as f:

            json.dump(
                {
                    "status": "idle",
                    "current": 0,
                    "total": 0,
                    "percent": 0
                },
                f,
                ensure_ascii=False
            )

    except Exception:
        pass

    return {
        "status": "ok"
    }

@router.get("/progress")
async def get_progress():

    # progressファイルが無い場合
    if not os.path.exists(PROGRESS_FILE):

        return {
            "current": 0,
            "total": 0,
            "percent": 0,
            "status": "idle"
        }

    try:

        with open(
            PROGRESS_FILE,
            "r",
            encoding="utf-8"
        ) as f:

            progress = json.load(f)

        return progress

    except Exception:

        return {
            "current": 0,
            "total": 0,
            "percent": 0,
            "status": "error"
        }

import json
import os
import threading

from uuid import uuid4

from fastapi import APIRouter
from fastapi import UploadFile
from fastapi import File
from fastapi import HTTPException

from fastapi.responses import FileResponse

from app.services.ocr_service import pdf_to_csv

router = APIRouter()

from app.config.settings import TEMP_DIR

UPLOAD_DIR = TEMP_DIR
PROGRESS_FILE = TEMP_DIR / "progress.json"

os.makedirs(
    UPLOAD_DIR,
    exist_ok=True
)

CURRENT_CSV = None


def run_ocr(
    pdf_path,
    csv_path
):
    global CURRENT_CSV

    try:

        result_csv = pdf_to_csv(
            pdf_path,
            csv_path
        )

        if os.path.exists(
            pdf_path
        ):
            os.remove(
                pdf_path
            )

        CURRENT_CSV = result_csv

    except Exception as e:

        with open(
            PROGRESS_FILE,
            "w",
            encoding="utf-8"
        ) as f:

            json.dump(
                {
                    "status": "error",
                    "current": 0,
                    "total": 0,
                    "percent": 0,
                    "message": str(e)
                },
                f,
                ensure_ascii=False
            )


@router.post("/upload")
async def upload_file(
    file: UploadFile = File(...)
):

    global CURRENT_CSV

    CURRENT_CSV = None

    with open(
        PROGRESS_FILE,
        "w",
        encoding="utf-8"
    ) as f:

        json.dump(
            {
                "status": "starting",
                "current": 0,
                "total": 0,
                "percent": 0
            },
            f,
            ensure_ascii=False
        )

    pdf_path = UPLOAD_DIR / f"{uuid4()}_{file.filename}"

    with open(
        pdf_path,
        "wb"
    ) as f:

        f.write(
            await file.read()
        )

    csv_path = pdf_path.with_suffix(".csv")

    thread = threading.Thread(
    target=run_ocr,
    args=(
        str(pdf_path),
        str(csv_path)
    ),
    daemon=True
)

    thread.start()

    return {
        "status": "started"
    }


@router.get("/download")
def download_csv():

    global CURRENT_CSV

    if not CURRENT_CSV:
        raise HTTPException(
            status_code=404,
            detail="CSV未生成"
        )

    return FileResponse(
        CURRENT_CSV,
        filename="result.csv",
        media_type="text/csv"
    )

def detect_company(text: str) -> str:
    """
    OCR/HTMLから抽出したテキストをもとに
    電力会社を判定する
    """

    # 東北電力
    if ("東北電力" in text) or ("ご使用量" in text and "東北" in text):
        return "東北電力"

    # 北陸電力
    if ("北陸電力" in text) or ("ほくリンク" in text):
        return "北陸電力"

    # 東京電力
    if ("東京電力" in text) or ("TEPCO" in text) or ("くらしTEPCO" in text):
        return "東京電力"

    # 中部電力
    if ("中部電力" in text) or ("カテエネ" in text):
        return "中部電力"

    # 判定できない場合
    return None

import csv
import uuid
import os

def generate_csv(data: dict) -> str:
    """
    抽出結果 dict を CSV に変換し、一時ファイルとして保存する
    FileResponse で返却できるようにファイルパスを返す
    """

    # 一時ファイルパス
    temp_id = str(uuid.uuid4())
    csv_path = f"/tmp/{temp_id}.csv"

    # CSVの列順（固定）
    headers = [
        "電力会社名",
        "契約種別",
        "検針日",
        "使用量(kWh)",
        "請求金額",
        "供給地点番号",
        "前月使用量",
        "燃料調整費",
        "再エネ賦課金",
        "基本料金",
        "従量料金",
        "電力量料金",
        "契約容量",
        "需要家番号",
        "請求月",
        "支払期限"
    ]

    # dict → CSV行に変換
    row = [
        data.get("company"),
        data.get("contract"),
        data.get("meter_date"),
        data.get("usage"),
        data.get("price"),
        data.get("supply_point"),
        data.get("prev_usage"),
        data.get("fuel_cost"),
        data.get("renewable_cost"),
        data.get("basic_fee"),
        data.get("meter_fee"),
        data.get("power_fee"),
        data.get("contract_amp"),
        data.get("customer_id"),
        data.get("billing_month"),
        data.get("due_date"),
    ]

    # CSV書き込み
    with open(csv_path, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(headers)
        writer.writerow(row)

    return csv_path

    result = {}

    result["company"] = "中部電力"

from bs4 import BeautifulSoup

def extract_text_from_html(html_path: str) -> str:
    """
    HTML（WEB版検針票）を DOM解析してテキスト抽出する
    """
    with open(html_path, "r", encoding="utf-8", errors="ignore") as f:
        html = f.read()

    soup = BeautifulSoup(html, "lxml")

    # HTML内のテキストをすべて抽出
    text = soup.get_text(separator="\n")

    return text

import pytesseract
from app.config.settings import POPPLER_PATH
import tempfile
import os

def extract_text_from_pdf(pdf_path: str) -> str:
    """
    紙PDF（画像ベース）を OCR してテキスト抽出する
    """
    text = ""

    # PDF → 画像化（300dpi）
    images = convert_from_path(
        pdf_path,
        poppler_path=str(POPPLER_PATH)
)

    for img in images:
        # 一時ファイルに画像保存
        with tempfile.NamedTemporaryFile(delete=False, suffix=".png") as tmp:
            img.save(tmp.name, "PNG")
            tmp_path = tmp.name

        # OCR実行（日本語）
        page_text = pytesseract.image_to_string(tmp_path, lang="jpn")
        text += page_text + "\n"

        # 一時ファイル削除
        os.remove(tmp_path)

    return text

def update_progress(
    current,
    total,
    status="running"
):

    percent = 0

            current / total * 100,

                "current": current,

def extract_text_horizontal(image):

    print("===== OCR START =====")
    print(
        "Tesseract Version:",
        pytesseract.get_tesseract_version()
    )

    data = pytesseract.image_to_data(
        image,
        lang="jpn",
        output_type=Output.DICT
    )

    rows = []

    count = len(
        data["text"]
    )

    for i in range(count):

        text = data["text"][i].strip()

        if not text:
            continue

        rows.append(
            {
                "text": text,
                "top": data["top"][i],
                "left": data["left"][i]
            }
        )

    rows.sort(
        key=lambda x: (
            round(x["top"] / 15),
            x["left"]
        )
    )

    return rows


def merge_rows(rows):

    merged_lines = []

    current_row = []
    current_top = None

    for row in rows:

        top_group = round(
            row["top"] / 15
        )

        if current_top is None:
            current_top = top_group

        if current_top != top_group:

            merged_lines.append(
                " ".join(current_row)
            )

            current_row = []
            current_top = top_group

        current_row.append(
            row["text"]
        )

    if current_row:

        merged_lines.append(
            " ".join(current_row)
        )

    return merged_lines


def pdf_to_csv(
    pdf_path,
    csv_path
):

    print("===== PDF INFO =====")

    info = pdfinfo_from_path(
    pdf_path,
    poppler_path=str(POPPLER_PATH)
)

    total_pages = info["Pages"]

    print(
        f"Total Pages: {total_pages}"
    )

    update_progress(
        0,
        total_pages
    )

    result_lines = []

    for page_no in range(
        1,
        total_pages + 1
    ):

        print(
            f"Page {page_no}"
        )

        update_progress(
            page_no,
            total_pages
        )

        print("convert start")

        images = convert_from_path(
            pdf_path,
            first_page=page_no,
            last_page=page_no,
            poppler_path=str(POPPLER_PATH)
        )
        print("convert done")

        image = images[0]

        rows = extract_text_horizontal(
            image
        )

        print("ocr done")

        lines = merge_rows(
            rows
        )

        result_lines.extend(
            lines
        )

    with open(
        csv_path,
        "w",
        newline="",
        encoding="utf-8-sig"
    ) as f:

        writer = csv.writer(f)

        writer.writerow(
            ["OCR結果"]
        )

        for line in result_lines:

            writer.writerow(
                [line]
            )

    update_progress(
        total_pages,
        total_pages,
        "completed"
    )

    return csv_path

/* Progress監視 */

function startProgressMonitor() {

    if (progressTimer) {

        clearInterval(
            progressTimer
        );
    }

    fetchErrorCount = 0;

    progressTimer =
        setInterval(
            async () => {

                try {

                    const response =
                        await fetch(
                            "/progress"
                        );

                    const data =
                        await response.json();

                    fetchErrorCount = 0;

                    errorBox.classList.add(
                        "hidden"
                    );

                    lastCurrent =
                        data.current ?? lastCurrent;

                    lastTotal =
                        data.total ?? lastTotal;

                    lastPercent =
                        data.percent ?? lastPercent;

                    progressText.innerText =
                        `${lastCurrent} / ${lastTotal}`;

                    percentText.innerText =
                        `${lastPercent}%`;

                    progressBar.style.width =
                        `${lastPercent}%`;

                    if (
                        data.status === "completed"
                    ) {

                        clearInterval(
                            progressTimer
                        );

                        loadingBox.classList.add(
                            "hidden"
                        );

                        // CSV自動ダウンロード

                        const a =
                            document.createElement(
                                "a"
                            );

                        a.href =
                            "/download";

                        document.body.appendChild(
                            a
                        );

                        a.click();

                        a.remove();

                        // 完了モーダル

                        completeModal.classList.remove(
                            "hidden"
                        );

                        return;
                    }

                    if (
                        data.status ===
                        "error"
                    ) {

                        clearInterval(
                            progressTimer
                        );

                        loadingBox.classList.add(
                            "hidden"
                        );

                        errorBox.innerText =
                            data.message ||
                            "OCRエラー";

                        errorBox.classList.remove(
                            "hidden"
                        );

                        return;
                    }

                }
                catch(error) {

                    fetchErrorCount++;

                    console.warn(
                        "progress fetch error",
                        fetchErrorCount
                    );

                    // 最後の進捗を保持
                    progressText.innerText =
                        `${lastCurrent} / ${lastTotal}`;

                        `${lastPercent}%`;

                            "進捗取得に失敗しています";

                        errorBox.classList.remove(
                            "hidden"
                        );
                    }
                }

            },
            1000
        );
}

HOST = "127.0.0.1"

    webbrowser.open(
        f"http://{HOST}:{PORT}"

def ensure_runtime_folders():

    base_dir = Path(__file__).resolve().parent

    (base_dir / "temp").mkdir(
        exist_ok=True
    )

    (base_dir / "logs").mkdir(
        exist_ok=True
    )

    ensure_runtime_folders()

    threading.Timer(
        2,
        open_browser
    ).start()

        host=HOST,
        port=PORT,
        reload=False,
        log_level="info"
    )

scanmeter-app
Version 1.0

=========================================
概要
=========================================

scanmeter-app は検針票PDFをOCR解析し、
CSVファイルへ変換するアプリです。

インターネット接続は不要です。

処理はすべてPC内で実行されます。


=========================================
対応内容
=========================================

・PDFアップロード
・OCR解析
・進捗表示
・CSV生成
・CSVダウンロード
・処理完了後自動クリーンアップ


=========================================
動作環境
=========================================

Windows 10
Windows 11

推奨メモリ

8GB以上


=========================================
起動方法
=========================================

scanmeter-app.exe

をダブルクリックしてください。

数秒後にブラウザが起動します。

起動しない場合は

http://127.0.0.1:8000

へアクセスしてください。


=========================================
利用手順
=========================================

① PDFを選択

② OCR開始を押す

③ 進捗を待つ

④ CSV出力完了

⑤ CSVを保存する


=========================================
処理中の注意
=========================================

OCR実行中は

・アプリを終了しない
・ブラウザを閉じない
・PCをシャットダウンしない

でください。

ページ数が多いPDFは
完了まで時間がかかります。

=========================================

scanmeter-app.exe
    アプリ本体

temp
    一時ファイル保存先

    ログ保存先

改善しない場合は

logs

フォルダ内のログを確認してください。
